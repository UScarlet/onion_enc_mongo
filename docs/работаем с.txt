
## Как выглядят структуры, с которыми работает программа

### до шифрования ()

parsed = {
    "header": (1001, 0, 2013),  # (request_id, response_to, op_code)
    "flags": 0,
    "checksum": None,
    
    # ВОТ ОН — СПИСОК СЕКЦИЙ! Внутри него лежат словари-секции
    "sections": [
        
        # Первая секция (Всегда Титульный лист, тип 0)
        {
            "type": 0,      # payload_type
            # Внутри "document" лежат байты BSON, которые содержат:
            # {"insert": "test", "$db": "test_proxy"}
            "document": b'\x28\x00\x00\x00\x02insert\x00...' 
        },
        
        # Вторая секция (Данные, тип 1 — присутствует при insert)
        {
            "type": 1,      # payload_type
            "identifier": "documents",
            # Внутри "documents" лежит список байтовых документов сотрудников
            "documents": [
                b'\x1e\x00\x00\x00\x02name\x00\x06\x00\x00\x00Alice\x00...', # Байты Элис
                b'\x1c\x00\x00\x00\x02name\x00\x04\x00\x00\x00Bob\x00...'    # Байты Боба
            ]
        }
        
    ]
}

### после шифрования

parsed = {
    "header": (1001, 0, 2013),  # (request_id, response_to, op_code сообщения OP_MSG)
    "flags": 0,                 # Системные флаги протокола MongoDB Wire Protocol
    "checksum": None,           # Поле для контрольной суммы пакета (если применимо)
    
    "sections": [
        
        # --- СЕКЦИЯ ТИПА 0 (Управляющая команда СУБД) ---
        {
            "type": 0,
            "document": {
                "insert": "test",        # Имя целевой коллекции, определенное динамически
                "$db": "test_proxy"      # Имя целевой базы данных, определенное динамически
            }
        },
        
        # --- СЕКЦИЯ ТИПА 1 (Криптографический конвейер данных) ---
        {
            "type": 1,
            "identifier": "documents",   # Маркер того, что внутри лежит массив документов СУБД
            "documents": [               # Список документов, прошедший инкапсуляцию слоев:
                
                # ------ ПЕРВЫЙ ДОКУМЕНТ (Сотрудник Alice) ------
                {
                    # Текстовое поле: защищено только детерминированным слоем (AES-GCM-SIV)
                    "name__det": "i:iUhb76Gz9Xwqa_dl...==",     
                    
                    # Числовое поле: оригинал переписан в детерминированный шифртекст равенства
                    "salary__det": "d:83jf9baK10pL_mz...==",    
                    
                    # Гомоморфный аддитивный слой (Пэйе): хранит ciphertext и exponent для СУБД агрегаций
                    "salary__hom": {                            
                        "ciphertext": "843f9ba023c71e98d8f4323ab...",
                        "exponent": -2
                    },
                    
                    # Слой сохранения порядка HOPE: хранит вероятностный математический слепок для сравнений
                    "salary__hope": {                           
                        "hope_cipher": "9238472938471029384729384..."
                    }
                },
                
                # ------ ВТОРОЙ ДОКУМЕНТ (Сотрудник Bob) ------
                {
                    "name__det": "i:pLkm98Wq11vXz_bn...==",
                    "salary__det": "d:72hd81mK92jH_qp...==",
                    "salary__hom": {
                        "ciphertext": "112a4f6d89e21b88aef315cd8...",
                        "exponent": -2
                    },
                    "salary__hope": {
                        "hope_cipher": "4920184729102837491029384..."
                    }
                }
                
            ]
        }
    ]
}

### find

parsed = {
    "header": (1002, 0, 2013),
    "flags": 0,
    "checksum": None,
    
    "sections": [
        # --- СЕКЦИЯ ТИПА 0 (Управляющая команда + Трансформированный фильтр) ---
        {
            "type": 0,
            "document": {
                "find": "test",          # Имя коллекции
                
                # ВМЕСТО ОТКРЫТОГО {"salary": {"$gt": 45000}} 
                # прокси-сервер инжектирует в СУБД гомоморфное лямбда-выражение:
                "filter": {
                    "$where": (
                        "function() {"
                        "  if (!this.salary__hope) return false;"
                        "  function modPow(b, e, m) { ... };" # Встроенный быстрый BigInt modPow
                        "  let c_A = BigInt(this.salary__hope.hope_cipher);"
                        "  let c_B_inv = BigInt(\"293847293847...\");" # Константа вычислена в прокси
                        "  let n = BigInt(\"3233...\");"
                        "  let n_sq = n * n;"
                        "  let ck0 = BigInt(\"9182391823...\");"       # Передаем ключ ck0 для гашения шума
                        "  let ck1 = BigInt(\"10293...\");"            # Передаем ключ ck1
                        "  let c_diff = (c_A * c_B_inv) % n_sq;"
                        "  let delta_mod = (modPow(c_diff, ck0, n_sq) - 1n) / n * ck1 % n;"
                        "  let delta = delta_mod > n/2n ? delta_mod - n : delta_mod;" # Функция SMOD
                        "  return delta > 0n;" # Оператор $gt превратился в проверку "знак Дельта > 0"
                        "}"
                    )
                },
                
                "$db": "test_proxy"      # Имя базы данных
            }
        }
    ]
}
